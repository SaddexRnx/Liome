cmake_minimum_required(VERSION 3.22.1)
project(localmind_llama)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE ON CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_VULKAN OFF CACHE BOOL "" FORCE)
set(GGML_METAL OFF CACHE BOOL "" FORCE)
set(GGML_OPENCL OFF CACHE BOOL "" FORCE)

add_subdirectory(llama.cpp)
add_library(localmind_llama SHARED localmind_jni.cpp)
target_include_directories(localmind_llama PRIVATE llama.cpp/include llama.cpp/ggml/include)
target_link_libraries(localmind_llama PRIVATE llama log android)
if(ANDROID_ABI STREQUAL "arm64-v8a")
    target_compile_options(localmind_llama PRIVATE -march=armv8-a+fp16)
endif()
